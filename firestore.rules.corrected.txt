rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isAuthenticated() {
      return request.auth != null;
    }

    function userTenantPath(uid) {
      return /databases/$(database)/documents/userTenants/$(uid);
    }

    function isMappedToTenant(tenantId) {
      return isAuthenticated()
        && exists(userTenantPath(request.auth.uid))
        && get(userTenantPath(request.auth.uid)).data.tenantId == tenantId
        && get(userTenantPath(request.auth.uid)).data.status == 'active';
    }

    function mappedRole(tenantId) {
      return isMappedToTenant(tenantId)
        ? get(userTenantPath(request.auth.uid)).data.role
        : null;
    }

    function hasValidRole(role) {
      return role is string
        && role in ['owner', 'admin', 'editor', 'viewer'];
    }

    // userTenants es la asignación autoritativa administrada por el servidor.
    function isActiveMember(tenantId) {
      return isMappedToTenant(tenantId)
        && hasValidRole(mappedRole(tenantId));
    }

    function hasRole(tenantId, role) {
      return isActiveMember(tenantId)
        && mappedRole(tenantId) == role;
    }

    function isNormalCatalog(catalogId) {
      return catalogId == 'normal';
    }

    function isSocialNetworkId(network) {
      return network is string
        && network in [
          'facebook',
          'instagram',
          'whatsapp',
          'tiktok',
          'youtube'
        ];
    }

    function isValidSocialUrl(network, url) {
      return url is string
        && url.size() <= 2048
        && (
          (network == 'facebook'
            && url.matches('(?i)^https?://([a-z0-9-]+\\.)*facebook\\.com(/.*|[?#].*)?$'))
          || (network == 'instagram'
            && url.matches('(?i)^https?://([a-z0-9-]+\\.)*instagram\\.com(/.*|[?#].*)?$'))
          || (network == 'whatsapp'
            && (url.matches('(?i)^https?://([a-z0-9-]+\\.)*whatsapp\\.com(/.*|[?#].*)?$')
              || url.matches('(?i)^https?://wa\\.me(/.*|[?#].*)?$')))
          || (network == 'tiktok'
            && url.matches('(?i)^https?://([a-z0-9-]+\\.)*tiktok\\.com(/.*|[?#].*)?$'))
          || (network == 'youtube'
            && (url.matches('(?i)^https?://([a-z0-9-]+\\.)*youtube\\.com(/.*|[?#].*)?$')
              || url.matches('(?i)^https?://youtu\\.be(/.*|[?#].*)?$')))
        );
    }

    function isValidSocialRecord(socialId, data) {
      return isSocialNetworkId(data.network)
        && (socialId == data.network || socialId == 'socials')
        && data.keys().hasOnly([
          'catalogId',
          'network',
          'url',
          'sortOrder',
          'status',
          'createdAt',
          'updatedAt'
        ])
        && data.keys().hasAll([
          'catalogId',
          'network',
          'url',
          'sortOrder',
          'status',
          'createdAt',
          'updatedAt'
        ])
        && isNormalCatalog(data.catalogId)
        && isValidSocialUrl(data.network, data.url)
        && data.sortOrder is int
        && data.sortOrder >= 1
        && data.status in ['active', 'inactive']
        && data.createdAt is timestamp
        && data.updatedAt is timestamp;
    }

    function socialDocumentPath(tenantId, socialId) {
      return /databases/$(database)/documents/tenants/$(tenantId)/socials/$(socialId);
    }

    function canReadSocialNetwork(tenantId, socialId) {
      let socialPath = socialDocumentPath(tenantId, socialId);
      return isActiveMember(tenantId)
        && (isSocialNetworkId(socialId) || socialId == 'socials')
        && (!exists(socialPath) || get(socialPath).data.catalogId == 'normal');
    }

    function socialNetworkIsAvailableAfter(
      tenantId,
      socialId,
      comparedId,
      network
    ) {
      return socialId == comparedId
        || !existsAfter(socialDocumentPath(tenantId, comparedId))
        || getAfter(socialDocumentPath(tenantId, comparedId)).data.network != network;
    }

    function socialNetworkIsUniqueAfter(tenantId, socialId, network) {
      return socialNetworkIsAvailableAfter(tenantId, socialId, 'facebook', network)
        && socialNetworkIsAvailableAfter(tenantId, socialId, 'instagram', network)
        && socialNetworkIsAvailableAfter(tenantId, socialId, 'whatsapp', network)
        && socialNetworkIsAvailableAfter(tenantId, socialId, 'tiktok', network)
        && socialNetworkIsAvailableAfter(tenantId, socialId, 'youtube', network)
        && socialNetworkIsAvailableAfter(tenantId, socialId, 'socials', network);
    }

    function socialOrderIsAvailableAfter(
      tenantId,
      socialId,
      comparedId,
      sortOrder
    ) {
      return socialId == comparedId
        || !existsAfter(socialDocumentPath(tenantId, comparedId))
        || getAfter(socialDocumentPath(tenantId, comparedId)).data.sortOrder != sortOrder;
    }

    function socialOrderIsUniqueAfter(tenantId, socialId, sortOrder) {
      return socialOrderIsAvailableAfter(tenantId, socialId, 'facebook', sortOrder)
        && socialOrderIsAvailableAfter(tenantId, socialId, 'instagram', sortOrder)
        && socialOrderIsAvailableAfter(tenantId, socialId, 'whatsapp', sortOrder)
        && socialOrderIsAvailableAfter(tenantId, socialId, 'tiktok', sortOrder)
        && socialOrderIsAvailableAfter(tenantId, socialId, 'youtube', sortOrder)
        && socialOrderIsAvailableAfter(tenantId, socialId, 'socials', sortOrder);
    }

    function canManage(tenantId) {
      return hasRole(tenantId, 'owner')
        || hasRole(tenantId, 'admin');
    }

    function canEdit(tenantId) {
      return canManage(tenantId)
        || hasRole(tenantId, 'editor');
    }

    match /userTenants/{uid} {
      allow get: if isAuthenticated() && request.auth.uid == uid;
      allow list, create, update, delete: if false;
    }

    match /tenants/{tenantId} {
      allow get: if isActiveMember(tenantId);
      allow list: if false;
      allow update: if canManage(tenantId)
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['name', 'logoUrl', 'updatedAt']);
      allow create, delete: if false;

      match /members/{uid} {
        allow get: if isAuthenticated() && (
          request.auth.uid == uid || canManage(tenantId)
        );
        allow list: if canManage(tenantId);
        allow create, update, delete: if canManage(tenantId);
      }

      match /catalogs/{catalogId} {
        allow read: if isNormalCatalog(catalogId)
          && isActiveMember(tenantId);
        allow create, update, delete: if isNormalCatalog(catalogId)
          && canEdit(tenantId);

        match /items/{itemId} {
          allow read: if isNormalCatalog(catalogId)
            && isActiveMember(tenantId);
          allow create, update, delete: if isNormalCatalog(catalogId)
            && canEdit(tenantId);
        }
      }

      match /products/{productId} {
        allow read: if isActiveMember(tenantId);
        allow create, update, delete: if canEdit(tenantId);
      }

      match /categories/{categoryId} {
        allow read: if isActiveMember(tenantId)
          && resource.data.catalogId == 'normal';
        allow create: if canEdit(tenantId)
          && request.resource.data.catalogId == 'normal';
        allow update: if canEdit(tenantId)
          && resource.data.catalogId == 'normal'
          && request.resource.data.catalogId == 'normal';
        allow delete: if canEdit(tenantId)
          && resource.data.catalogId == 'normal';
      }

      match /socials/{socialId} {
        allow get: if canReadSocialNetwork(tenantId, socialId);
        allow list: if isActiveMember(tenantId)
          && resource.data.catalogId == 'normal';
        allow create: if canEdit(tenantId)
          && isValidSocialRecord(socialId, request.resource.data)
          && socialId == request.resource.data.network
          && socialNetworkIsUniqueAfter(
            tenantId,
            socialId,
            request.resource.data.network
          )
          && socialOrderIsUniqueAfter(
            tenantId,
            socialId,
            request.resource.data.sortOrder
          )
          && request.resource.data.createdAt == request.time
          && request.resource.data.updatedAt == request.time;
        allow update: if canEdit(tenantId)
          && isValidSocialRecord(socialId, request.resource.data)
          && socialNetworkIsUniqueAfter(
            tenantId,
            socialId,
            request.resource.data.network
          )
          && socialOrderIsUniqueAfter(
            tenantId,
            socialId,
            request.resource.data.sortOrder
          )
          && request.resource.data.createdAt == resource.data.createdAt
          && request.resource.data.updatedAt == request.time;
        allow delete: if canEdit(tenantId)
          && isValidSocialRecord(socialId, resource.data);
      }

      match /deliveryZones/{zoneId} {
        allow read: if isActiveMember(tenantId)
          && resource.data.catalogId == 'normal';
        allow create: if canEdit(tenantId)
          && request.resource.data.catalogId == 'normal';
        allow update: if canEdit(tenantId)
          && resource.data.catalogId == 'normal'
          && request.resource.data.catalogId == 'normal';
        allow delete: if canEdit(tenantId)
          && resource.data.catalogId == 'normal';
      }

      match /settings/{settingId} {
        allow read: if isActiveMember(tenantId);
        allow create, update, delete: if canManage(tenantId);
      }
    }

    match /plans/{planId} {
      allow read: if isAuthenticated();
      allow write: if false;
    }

    match /subscriptions/{subscriptionId} {
      allow read: if isAuthenticated()
        && resource.data.tenantId is string
        && isActiveMember(resource.data.tenantId);
      allow write: if false;
    }

    match /platformUsers/{uid} {
      allow get: if isAuthenticated() && request.auth.uid == uid;
      allow list, create, update, delete: if false;
    }
  }
}
